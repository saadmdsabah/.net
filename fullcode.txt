            View(await _db.Categories.AsNoTracking().ToListAsync());

        {

        {
            if (id==null) return NotFound();
            var cat = await _db.Categories.FindAsync(id);
if (cat == null)
    return NotFound();
return View(cat);
        }
 
        [HttpPost,ValidateAntiForgeryToken]
        public async Task<IActionResult> Edit(int id, Category obj)
        {
            if (id!=obj.Id) return NotFound();

            _db.Update(obj); await _db.SaveChangesAsync();

//UserController
 
using System.Linq;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
using Microsoft.AspNetCore.Identity;
using FastBite.Data;
using FastBite.Models;
using Microsoft.AspNetCore.Authorization;
 
namespace FastBite.Areas.Admin.Controllers
{
    [Area("Admin"), Authorize(Roles = "ADMIN")]
    public class UserController : Controller
    {
        private readonly ApplicationDbContext _db;
        private readonly UserManager<ApplicationUser> _um;
        public UserController(ApplicationDbContext db, UserManager<ApplicationUser> um)
        {
            _db = db; _um = um;
        }
 
        public async Task<IActionResult> Index() =>
            View(await _db.Users.AsNoTracking().ToListAsync());
 
        /* toggle lock / unlock ----------------------------- */
        public async Task<IActionResult> LockUnlock(string id)
        {
            var user = await _um.FindByIdAsync(id);
            if (user == null) return NotFound();
 
            if (user.LockoutEnd != null && user.LockoutEnd > System.DateTime.Now)
                user.LockoutEnd = System.DateTime.Now;              // unlock
            else
                user.LockoutEnd = System.DateTime.Now.AddYears(100); // lock
 
            await _um.UpdateAsync(user);
            return RedirectToAction(nameof(Index));
        }
    }
}

/* CartController.cs (Customer area) */
using System.Linq;
using System.Security.Claims;
using System.Threading.Tasks;
using FastBite.Data;
using FastBite.Models;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;
 
namespace FastBite.Areas.Customer.Controllers
{
    [Area("Customer")]
    public class CartController : Controller
    {
        private readonly ApplicationDbContext _db;
        public CartController(ApplicationDbContext db)=>_db=db;
        public IActionResult Index()
{
    return RedirectToAction(nameof(PlaceOrder));
}
        private async Task<Cart> GetCart()
        {
            var uid = User.FindFirstValue(ClaimTypes.NameIdentifier);
            var cart = await _db.Carts
                     .Include(c=>c.ApplicationUser)
                     .FirstOrDefaultAsync(c=>c.userId==uid && c.orderStatus==null);
            if(cart==null){
                cart=new Cart{userId=uid,OrderTotal=0,AmountWithoutDiscount=0};
                _db.Carts.Add(cart); await _db.SaveChangesAsync();
            }
            return cart;
        }
 
        public async Task<IActionResult> PlaceOrder()
        {
            var cart = await GetCart();
            var items = await _db.CartItems
                        .Include(ci=>ci.MenuItem)
                        .Where(ci=>ci.ApplicationUserId==cart.userId)
                        .ToListAsync();
 
            ViewBag.Items = items;
            ViewBag.Total = cart.OrderTotal.ToString();
            return View(cart);
        }
 
        [HttpPost]
        public async Task<IActionResult> ApplyCoupon(string offercode)
        {
            var cart = await GetCart();
            cart.offercode = offercode;
            var offer = await _db.Offers.FirstOrDefaultAsync(o=>o.Name==offercode && o.isActive);
            if(offer!=null && cart.AmountWithoutDiscount>=offer.MinimumAmount)
            {
                cart.discount = offer.Discount;
                cart.OrderTotal = cart.AmountWithoutDiscount - offer.Discount;
            }
            await _db.SaveChangesAsync();
            return RedirectToAction(nameof(PlaceOrder));
        }
 
        [HttpPost]
        public async Task<IActionResult> FinalPlace()
        {
            var cart = await GetCart();
            cart.orderStatus = "Order Placed and Pending Confirmation at Restaurant";
            await _db.SaveChangesAsync();
            return View("~/Views/Customer/Order/ConfirmOrder.cshtml", cart);
        }
        // …existing CartController code above …
 
[HttpPost]
public async Task<IActionResult> RemoveCoupon()
{
    var cart = await GetCart();
    cart.offercode = null;
    cart.discount  = 0;
    cart.OrderTotal = cart.AmountWithoutDiscount;
    await _db.SaveChangesAsync();
    return RedirectToAction(nameof(PlaceOrder));
}
 
/* plus / minus buttons */
public async Task<IActionResult> ChangeQty(int cartItemId, int delta)
{
    var item = await _db.CartItems.Include(ci=>ci.MenuItem).FirstOrDefaultAsync(ci=>ci.Id==cartItemId);
    if(item!=null)
    {
        item.Count = System.Math.Max(1, item.Count + delta);
        await _db.SaveChangesAsync();
 
        /* recalc */
var cart = await _db.Carts.FirstAsync(c => c.userId == item.ApplicationUserId && c.orderStatus == null);
        var uid  = item.ApplicationUserId;
        cart.AmountWithoutDiscount = await _db.CartItems
            .Where(ci=>ci.ApplicationUserId==uid)
            .SumAsync(ci=>ci.Count * ci.MenuItem.price);
        cart.OrderTotal = cart.AmountWithoutDiscount - cart.discount;
        await _db.SaveChangesAsync();
    }
    return RedirectToAction("PlaceOrder");
}
 
    }
}

public async Task<IActionResult> AddToCart(int menuItemId, int Count)
{
var uid = User.FindFirst(System.Security.Claims.ClaimTypes.NameIdentifier)?.Value;
if (string.IsNullOrEmpty(uid))
return RedirectToAction("Index", "Home", new { area = "Customer" });

if (cart == null)

}
 
var menu = await _db.MenuItems.Include(m => m.Restaurant).FirstOrDefaultAsync(m => m.Id == menuItemId);
if (menu == null)
{
// MenuItem not found, skip logic to prevent crash
return RedirectToAction("Index", "Home", new { area = "Customer" });
}
 
var firstItem = await _db.CartItems.Include(ci => ci.MenuItem)
.FirstOrDefaultAsync(ci => ci.ApplicationUserId == uid);
 
if (firstItem != null && firstItem.MenuItem != null && firstItem.MenuItem.RestaurantId != menu.RestaurantId)
{
_db.CartItems.RemoveRange(_db.CartItems.Where(ci => ci.ApplicationUserId == uid));
cart.AmountWithoutDiscount = cart.OrderTotal = 0;
}
 
var existing = await _db.CartItems
.FirstOrDefaultAsync(ci => ci.ApplicationUserId == uid && ci.MenuItemId == menuItemId);
if (existing == null)
{
existing = new CartItem
{
ApplicationUserId = uid,
MenuItemId = menuItemId,
Count = Count
};
_db.CartItems.Add(existing);
}
else
{
existing.Count += Count;
}
 
await _db.SaveChangesAsync();
 
// Safely calculate totals
var cartItems = await _db.CartItems
.Include(ci => ci.MenuItem)
.Where(ci => ci.ApplicationUserId == uid)
.ToListAsync();
 
cart.AmountWithoutDiscount = cartItems.Sum(ci => ci.Count * (ci.MenuItem?.price ?? 0));
cart.OrderTotal = cart.AmountWithoutDiscount - cart.discount;
await _db.SaveChangesAsync();
 
return RedirectToAction("MenuItem", "CartItem", new { area = "Customer", menuItemId = menuItemId, count = Count });
}

        public async Task<IActionResult> OrderHistory()
        {
            var list = await _db.Carts
                        .Where(c=>c.orderStatus!=null)
                        .ToListAsync();
            return View(list);     // cards with class="card"
        }
 
        public async Task<IActionResult> PendingRestaurantOrderDetails(int restId)
        {
            var carts = await _db.Carts
                        .Include(c=>c.ApplicationUser)
                        .Include(c=>c.CartItems)
                        .ThenInclude(ci=>ci.MenuItem)
                        .Where(c=>c.CartItems.Any(ci=>ci.MenuItem.RestaurantId==restId)
                               && (c.orderStatus=="Order Placed and Pending Confirmation at Restaurant"
                                   || c.orderStatus=="Order Approved and Food is being Prepared"))
                        .ToListAsync();
            ViewBag.RestId = restId;
            return View(carts);
        }
 
        /* links clicked in the view */
        public async Task<IActionResult> AcceptOrder(int id,int restId)
        {
            var cart=await _db.Carts.FindAsync(id);
            cart.orderStatus="Order Approved and Food is being Prepared";
            await _db.SaveChangesAsync();
            return RedirectToAction(nameof(PendingRestaurantOrderDetails),new{restId});
        }
        public async Task<IActionResult> Prepared(int id,int restId)
        {
            var c=await _db.Carts.FindAsync(id);
            c.orderStatus="Order Ready for Pickup";
            await _db.SaveChangesAsync();
            return RedirectToAction(nameof(PendingRestaurantOrderDetails),new{restId});
        }
        public async Task<IActionResult> Cancel(int id,int restId)
        {
            var c=await _db.Carts.FindAsync(id);
            c.orderStatus="Order Cancelled";
            await _db.SaveChangesAsync();
            return RedirectToAction(nameof(PendingRestaurantOrderDetails),new{restId});
